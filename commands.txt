------------------------------------------------------------------------------
 1. SETUP  -  install dependencies (run once)
------------------------------------------------------------------------------
pip install -r requirements.txt


------------------------------------------------------------------------------
 2. RUN THE UNIT TESTS  -  verify the solver before training
------------------------------------------------------------------------------
python -m pytest tests/ -v

# Run a single test file:
python -m pytest tests/test_masks.py -v


------------------------------------------------------------------------------
 3. TRAIN THE MODEL
------------------------------------------------------------------------------
# Full training with the default config:
python main_train.py --config configs/default.yaml

# Useful overrides:
python main_train.py --config configs/default.yaml --epochs 50
python main_train.py --config configs/default.yaml --device cpu
python main_train.py --config configs/default.yaml --seed 123

# TIP: to iterate fast, first train only on the small instances by editing
# configs/default.yaml:
#     dataset:
#       train_size_filter: ["C5"]
#       eval_size_filter: ["C5"]
# then scale up to the full 100-customer instances (set both filters to null).


------------------------------------------------------------------------------
 3b. CURRICULUM TRAINING  (Phase 0 checks + phased C -> C+R -> C+R+RC -> hard)
------------------------------------------------------------------------------
# Run ONLY the Phase 0 environment & mask sanity checks (no training):
python main_train.py --config configs/curriculum.yaml --phase0-only

# Run the full 250-epoch curriculum (Phase 0 must pass first):
#   Phase 1:  50 epochs on C
#   Phase 2:  70 epochs on C + R        (50/50 balanced batches)
#   Phase 3: 100 epochs on C + R + RC   (1/3 each, per-class validation)
#   Phase 4:  30 epochs on hard cases   (50% hard + 50% normal)
python main_train.py --config configs/curriculum.yaml --curriculum

# (Equivalent: a config with a 'curriculum:' block auto-runs curriculum mode.)
python main_train.py --config configs/curriculum.yaml

# Train anyway even if Phase 0 fails (NOT recommended):
python main_train.py --config configs/curriculum.yaml --curriculum --force

# Checkpoints saved by the curriculum:
#   best.pt / best_overall.pt, best_C.pt, best_R.pt, best_RC.pt,
#   best_feasibility.pt, last.pt
# Validation score used for "best": cost + 10000 * (1 - feasibility_rate)


------------------------------------------------------------------------------
 4. LIVE TRAINING DASHBOARD  (run in a SECOND terminal, while training runs)
------------------------------------------------------------------------------
python main_dashboard.py --config configs/default.yaml

# Then open this in your browser:
#     http://localhost:8080
# The page auto-refreshes every 3 seconds with all metrics.

# If port 8080 is busy, choose another port:
python main_dashboard.py --config configs/default.yaml --port 8090

# Point the dashboard at a specific log file:
python main_dashboard.py --log outputs/logs/train_log.jsonl --port 8090


------------------------------------------------------------------------------
 5. EVALUATE THE TRAINED MODEL  (model + baselines + fleet-size search)
------------------------------------------------------------------------------
python main_evaluate.py --config configs/default.yaml

# Evaluate a specific checkpoint:
python main_evaluate.py --config configs/default.yaml --checkpoint outputs/checkpoints/best.pt

# Force CPU:
python main_evaluate.py --config configs/default.yaml --device cpu


------------------------------------------------------------------------------
 6. WHERE THE OUTPUTS GO
------------------------------------------------------------------------------
# Checkpoints : outputs/checkpoints/   (best.pt, last.pt, epoch_*.pt; curriculum adds best_C/R/RC/feasibility.pt)
# Train log   : outputs/logs/train_log.jsonl   (read by the dashboard)
# Val log     : outputs/logs/val_log.jsonl     (per-class validation, curriculum)
# Results CSV : outputs/results/results.csv, training_metrics.csv, validation_metrics.csv
# Phase 0     : outputs/results/phase0_report.txt
# Summaries   : outputs/results/curriculum_summary.json, training_summary.json
# Routes JSON : outputs/routes/<instance>_routes.json
# Figures     : outputs/figures/training/   (curves, overview, composition, validation/ per-class)
#               outputs/figures/            (fleet-search, routes, comparisons)
# Run archive : outputs/runs/<run_id>/      (config, metrics, validation, summary)


------------------------------------------------------------------------------
 7. TYPICAL FULL WORKFLOW
------------------------------------------------------------------------------
# Terminal 1:
pip install -r requirements.txt
python -m pytest tests/ -v
python main_train.py --config configs/default.yaml

# Terminal 2 (started right after training begins):
python main_dashboard.py --config configs/default.yaml
#   -> open http://localhost:8080

# After training finishes:
python main_evaluate.py --config configs/default.yaml
